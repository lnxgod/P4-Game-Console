cmake_minimum_required(VERSION 3.16)
project(console_cartridge_timing LANGUAGES C)
include(CTest)
add_executable(cartridge_timing_tests test_cartridge_timing.c)
target_include_directories(cartridge_timing_tests PRIVATE ../main)
target_compile_features(cartridge_timing_tests PRIVATE c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(cartridge_timing_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(cartridge_timing_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME cartridge_timing_tests COMMAND cartridge_timing_tests)
