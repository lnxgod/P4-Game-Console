cmake_minimum_required(VERSION 3.19)
project(wacky_p4_probe C)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(PORT "${ROOT}/.tools/wacky-p4-port")
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL3 REQUIRED IMPORTED_TARGET sdl3)
add_subdirectory("${ROOT}/components/p4_game_api" p4_game_api)
add_subdirectory("${ROOT}/components/p4_signal_scan" p4_signal_scan)
add_subdirectory("${ROOT}/components/p4_game_save" p4_game_save)
file(GLOB CORE "${PORT}/ww_*.c")
add_subdirectory("${ROOT}/components/p4_midi" p4_midi)
target_compile_options(p4_midi PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
add_library(wacky_core STATIC ${CORE})
target_include_directories(wacky_core PUBLIC "${PORT}" "${ROOT}/components/p4_midi/include" "${ROOT}/components/p4_game_api/include")
target_compile_options(wacky_core PRIVATE -Wall -Wextra -Werror=implicit-function-declaration
    -fsanitize=address,undefined -fno-omit-frame-pointer -include "${PORT}/port_support.h")
target_compile_definitions(wacky_core PRIVATE malloc=ww_p4_malloc calloc=ww_p4_calloc free=ww_p4_free strlen=ww_p4_strlen strncpy=ww_p4_strncpy strcpy=ww_p4_strcpy strtol=ww_p4_strtol toupper=ww_p4_toupper snprintf=ww_p4_snprintf abs=ww_p4_abs)
add_library(wacky_port STATIC "${PORT}/port.c" "${PORT}/network.c" "${PORT}/port_support.c" "${PORT}/portable_libc.c" "${PORT}/sound.c" "${PORT}/front.c" "${PORT}/controls.c")
target_compile_definitions(wacky_port PRIVATE WW_P4_PROBE_TEST)
target_link_libraries(wacky_port PUBLIC wacky_core p4_game_api p4_midi)
target_compile_options(wacky_port PRIVATE -Wall -Wextra -Wconversion -Wshadow -Werror
    -fsanitize=address,undefined -fno-omit-frame-pointer)
# Compile the existing generic SDL3 runner unchanged against the actual port.
add_executable(wacky_play "${ROOT}/tools/p4-game-host/play_game.c"
    "${ROOT}/tools/p4-game-host/host_service.c")
set_target_properties(wacky_play PROPERTIES MACOSX_BUNDLE TRUE MACOSX_BUNDLE_GUI_IDENTIFIER "org.p4console.wacky-probe" MACOSX_BUNDLE_BUNDLE_NAME "Wacky Wheels Probe")
target_link_libraries(wacky_play PRIVATE wacky_port p4_game_save p4_signal_scan PkgConfig::SDL3)
target_compile_definitions(wacky_play PRIVATE P4_HOST_RENDER_DIVISOR=1 P4_HOST_GAME_DESCRIPTOR=p4_wacky_probe_game
    "P4_HOST_RESOURCE_PATH=\"${ROOT}/.tools/wacky-shareware/WACKY.DAT\"")
target_compile_options(wacky_play PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(wacky_play PRIVATE -fsanitize=address,undefined)
add_executable(wacky_checks probe_checks.c)
target_link_libraries(wacky_checks PRIVATE wacky_port)
target_compile_definitions(wacky_checks PRIVATE "WW_DATA_PATH=\"${ROOT}/.tools/wacky-shareware/WACKY.DAT\"")
target_compile_options(wacky_checks PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(wacky_checks PRIVATE -fsanitize=address,undefined)
include(CTest)
add_executable(wacky_touch_checks touch_checks.c)
target_link_libraries(wacky_touch_checks PRIVATE wacky_port)

add_test(NAME wacky_play_smoke COMMAND wacky_play --frames 240)
set_tests_properties(wacky_play_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy;UBSAN_OPTIONS=halt_on_error=1")
add_test(NAME wacky_port_checks COMMAND wacky_checks)
set_tests_properties(wacky_port_checks PROPERTIES ENVIRONMENT "UBSAN_OPTIONS=halt_on_error=1")
add_executable(wacky_validate_elf validate_elf.c "${ROOT}/components/p4_game_package/src/game_package.c" "${ROOT}/components/p4_multiplayer_registry/src/multiplayer_registry.c")
target_include_directories(wacky_validate_elf PRIVATE "${ROOT}/components/p4_game_package/include" "${ROOT}/components/p4_multiplayer_registry/include" "${ROOT}/components/p4_multiplayer/include")
target_link_libraries(wacky_validate_elf PRIVATE p4_game_api)
target_compile_options(wacky_validate_elf PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined)

target_compile_definitions(wacky_full_checks PRIVATE "WW_DATA_PATH=\"${ROOT}/.tools/wacky-shareware/WACKY.DAT\"")
