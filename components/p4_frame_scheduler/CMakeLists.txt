if(COMMAND idf_component_register)
    idf_component_register(SRCS "src/frame_scheduler.c" INCLUDE_DIRS "include")
    target_compile_features(${COMPONENT_LIB} PUBLIC c_std_11)
    target_compile_options(${COMPONENT_LIB} PRIVATE -Wall -Wextra -Wconversion -Wshadow -Werror)
else()
    cmake_minimum_required(VERSION 3.16)
    project(p4_frame_scheduler LANGUAGES C)
    add_library(p4_frame_scheduler STATIC src/frame_scheduler.c)
    target_include_directories(p4_frame_scheduler PUBLIC include)
    target_compile_features(p4_frame_scheduler PUBLIC c_std_11)
    if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(p4_frame_scheduler PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(p4_frame_scheduler PUBLIC -fsanitize=address,undefined)
    endif()
    include(CTest)
    if(BUILD_TESTING)
        add_executable(p4_frame_scheduler_tests tests/test_frame_scheduler.c)
        target_link_libraries(p4_frame_scheduler_tests PRIVATE p4_frame_scheduler)
        target_compile_options(p4_frame_scheduler_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
        add_test(NAME p4_frame_scheduler COMMAND p4_frame_scheduler_tests)
    endif()
endif()
