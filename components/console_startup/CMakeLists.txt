if(COMMAND idf_component_register)
    idf_component_register(SRCS src/startup.c src/fanfare.c src/flight.c INCLUDE_DIRS include "../console_shell/include" REQUIRES p4_cp437)
    target_compile_options(${COMPONENT_LIB} PRIVATE -Wall -Wextra -Wconversion -Wshadow -Werror)
else()
    cmake_minimum_required(VERSION 3.16)
    project(console_startup LANGUAGES C)
    if(NOT TARGET p4_cp437)
        add_subdirectory(../p4_cp437 "${CMAKE_CURRENT_BINARY_DIR}/p4_cp437")
    endif()
    add_library(console_startup STATIC src/startup.c src/fanfare.c src/flight.c)
    target_include_directories(console_startup PUBLIC include "../console_shell/include")
    target_link_libraries(console_startup PUBLIC p4_cp437 m)
    target_compile_options(console_startup PRIVATE -Wall -Wextra -Wconversion -Wshadow -Werror -fsanitize=address,undefined)
    target_link_options(console_startup PUBLIC -fsanitize=address,undefined)
    include(CTest)
    add_executable(console_startup_flight_preview tests/flight_preview.c)
    target_link_libraries(console_startup_flight_preview PRIVATE console_startup)
    add_test(NAME console_startup_flight_validation COMMAND console_startup_flight_preview
        "${CMAKE_CURRENT_LIST_DIR}/../../apps/console_os/main/assets/gamechangers_mark_flight.rgb565a8")
    add_executable(console_startup_preview tests/preview.c)
    target_link_libraries(console_startup_preview PRIVATE console_startup)
    add_test(NAME console_startup_validation COMMAND console_startup_preview
        "${CMAKE_CURRENT_LIST_DIR}/../../apps/console_os/main/assets/gamechangers_mark_v042.rgb565a8"
        "${CMAKE_CURRENT_BINARY_DIR}/boot.ppm" "${CMAKE_CURRENT_BINARY_DIR}/fanfare.wav")
endif()
