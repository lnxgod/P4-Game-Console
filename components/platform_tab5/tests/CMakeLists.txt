cmake_minimum_required(VERSION 3.16)
project(tab5_board_tests C)
enable_testing()
add_executable(tab5_speaker_test test_speaker.c ../src/tab5.c)
target_include_directories(tab5_speaker_test PRIVATE mocks ../include)
target_compile_definitions(tab5_speaker_test PRIVATE CONFIG_P4_BOARD_M5STACK_TAB5=1)
target_compile_features(tab5_speaker_test PRIVATE c_std_11)
target_compile_options(tab5_speaker_test PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(tab5_speaker_test PRIVATE -fsanitize=address,undefined)
add_test(NAME tab5_speaker_readback COMMAND tab5_speaker_test)

add_executable(tab5_sensor_test test_sensors.c ../src/sensor_decode.c)
target_include_directories(tab5_sensor_test PRIVATE ../include)
target_compile_features(tab5_sensor_test PRIVATE c_std_11)
target_compile_options(tab5_sensor_test PRIVATE -Wall -Wextra -Wconversion -Wshadow -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(tab5_sensor_test PRIVATE -fsanitize=address,undefined)
add_test(NAME tab5_sensor_decode COMMAND tab5_sensor_test)

add_test(NAME tab5_startup_retry COMMAND tab5_speaker_test retry)
add_test(NAME tab5_startup_recreate COMMAND tab5_speaker_test recreate)
add_test(NAME tab5_startup_bounded COMMAND tab5_speaker_test bounded)
set_tests_properties(tab5_speaker_readback tab5_sensor_decode tab5_startup_retry tab5_startup_recreate tab5_startup_bounded PROPERTIES TIMEOUT 15)

add_executable(tab5_rtc_clock_test test_rtc_clock.c ../src/rtc_clock.c ../src/sensor_decode.c)
target_include_directories(tab5_rtc_clock_test PRIVATE mocks ../include)

target_link_options(tab5_rtc_clock_test PRIVATE -fsanitize=address,undefined)
