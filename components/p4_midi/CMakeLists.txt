cmake_minimum_required(VERSION 3.19)
if(COMMAND idf_component_register)
    idf_component_register(SRCS "src/midi.c" INCLUDE_DIRS "include")
else()
    project(p4_midi C)
    add_library(p4_midi STATIC src/midi.c)
    target_include_directories(p4_midi PUBLIC include)
    target_compile_options(p4_midi PRIVATE -Wall -Wextra -Werror)
    if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
        enable_testing()
        target_compile_options(p4_midi PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(p4_midi INTERFACE -fsanitize=address,undefined)
        add_executable(test_midi tests/test_midi.c)
        target_link_libraries(test_midi PRIVATE p4_midi)
        target_compile_options(test_midi PRIVATE -fsanitize=address,undefined)
        add_test(NAME midi COMMAND test_midi)
    endif()
endif()
