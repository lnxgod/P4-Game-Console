cmake_minimum_required(VERSION 3.16)
project(p4_wifi_tests C)
enable_testing()
add_executable(wifi_test test_wifi.c ../src/discovery.c ../../p4_multiplayer/src/packet.c)
target_include_directories(wifi_test PRIVATE mocks ../include ../../platform_ble_host/include ../../p4_multiplayer/include ../../p4_game_api/include)
target_compile_options(wifi_test PRIVATE -Wall -Wextra -Wconversion -Wshadow -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(wifi_test PRIVATE -fsanitize=address,undefined)
add_test(NAME standalone_wifi COMMAND wifi_test)
set_tests_properties(standalone_wifi PROPERTIES TIMEOUT 15)
