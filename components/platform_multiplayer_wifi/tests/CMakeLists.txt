cmake_minimum_required(VERSION 3.16)
project(p4_wifi_tests C)
enable_testing()
option(P4_TEST_ASSUME_NO_DRAIN_HINT "Run the missing-hint behavioral baseline" OFF)
add_executable(wifi_poll_drained_test test_poll_drained.c ../src/discovery.c ../../p4_multiplayer/src/packet.c)
target_include_directories(wifi_poll_drained_test PRIVATE mocks ../include ../../platform_ble_host/include ../../p4_multiplayer/include)
target_compile_options(wifi_poll_drained_test PRIVATE -Wall -Wextra -Wconversion -Wshadow -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(wifi_poll_drained_test PRIVATE -fsanitize=address,undefined)
if(P4_TEST_ASSUME_NO_DRAIN_HINT)
    target_compile_definitions(wifi_poll_drained_test PRIVATE P4_TEST_ASSUME_NO_DRAIN_HINT=1)
endif()
add_test(NAME wifi_poll_drained COMMAND wifi_poll_drained_test)
set_tests_properties(wifi_poll_drained PROPERTIES TIMEOUT 15)
add_executable(wifi_test test_wifi.c ../src/discovery.c ../../p4_multiplayer/src/packet.c)
target_include_directories(wifi_test PRIVATE mocks ../include ../../platform_ble_host/include ../../p4_multiplayer/include ../../p4_game_api/include)
target_compile_options(wifi_test PRIVATE -Wall -Wextra -Wconversion -Wshadow -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(wifi_test PRIVATE -fsanitize=address,undefined)
add_test(NAME standalone_wifi COMMAND wifi_test)
set_tests_properties(standalone_wifi PROPERTIES TIMEOUT 15)
add_executable(wifi_retry_test test_retry.c ../src/discovery.c ../../p4_multiplayer/src/packet.c)
target_include_directories(wifi_retry_test PRIVATE mocks ../include ../../platform_ble_host/include ../../p4_multiplayer/include ../../p4_game_api/include)
target_compile_options(wifi_retry_test PRIVATE -Wall -Wextra -Wconversion -Wshadow -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(wifi_retry_test PRIVATE -fsanitize=address,undefined)
foreach(step RANGE 0 7)
    add_test(NAME wifi_retry_step_${step} COMMAND wifi_retry_test retry ${step})
    add_test(NAME wifi_cancel_step_${step} COMMAND wifi_retry_test cancel ${step})
    set_tests_properties(wifi_retry_step_${step} wifi_cancel_step_${step} PROPERTIES TIMEOUT 15)
endforeach()
add_test(NAME wifi_cancel_radio_wait COMMAND wifi_retry_test cancel -1)
set_tests_properties(wifi_cancel_radio_wait PROPERTIES TIMEOUT 15)
foreach(case lost-init-reply init-not-applied init-query-error init-query-cancel)
    add_test(NAME wifi_${case} COMMAND wifi_retry_test ${case})
    set_tests_properties(wifi_${case} PROPERTIES TIMEOUT 15)
endforeach()
