cmake_minimum_required(VERSION 3.16)
project(console_settings_tests LANGUAGES C)
add_executable(console_settings_tests test_settings.c ../src/platform_console_settings.c ../../gamepad_core/src/mapping.c)
target_include_directories(console_settings_tests PRIVATE stubs ../include ../../gamepad_core/include)
target_compile_definitions(console_settings_tests PRIVATE CONFIG_P4_BOARD_M5STACK_TAB5=1)

target_link_options(console_settings_tests PRIVATE -fsanitize=address,undefined)
enable_testing()
add_test(NAME console_settings_tests COMMAND console_settings_tests)
